# Protocol 1.1: a host LIST (the HOME double-tap mode list), picked and cancelled; a 1.0 host never gets one.
@ACK@
wait 100
expect host_minor 1
@TAPE@
wait 50
sysex_out
@LIST@
wait 100
expect list 2
shot list.png
# the buttons stay local while the list is open: PLAY and ENV do not reach the host
turn SELECT 1
wait 100
expect list 3
turn SELECT -2
wait 100
expect list 1
turn SELECT 1
wait 100
expect list 2
# HOME picks: LIST_PICK index 1 (POLY), action OK
tap HOME
wait 100
expect list 0
expect_sysex @PICK1_0@
sysex_out
# a list again: ENV cancels (LIST_PICK, action CANCEL); PLAY / HOME presses never went out as BTN
@LIST@
wait 50
tap ENV
wait 100
expect list 0
expect_sysex @PICK1_1@
sysex_out
