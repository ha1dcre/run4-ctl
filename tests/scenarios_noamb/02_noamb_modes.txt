# Protocol 1.4 in the package without AMB and DUB mode: MODES lists them as not available, switching to them is refused (NOMODE),
# SYNTH works and is described.
@ACK14@
wait 100
expect_sysex @MODE_RUN4@
sysex_out
@MODES_REQ@
wait 50
expect_sysex @MODES_NOAMB_0@
@MODE_SET_AMB@
wait 100
expect_sysex @ERR_NOMODE_RUN4@
expect mode 0
@MODE_SET_DUB@
wait 100
expect_sysex @ERR_NOMODE_RUN4@
expect mode 0
sysex_log out:lm_noamb_synth.bin
@MODE_SET_SYNTH@
wait 200
expect mode 1
expect connected 1
@DESC_REQ_SYNTH@
wait 1500
expect lm_stream 0
expect lm_nparams 143
sysex_log off
sysex_out
@MODE_SET_AMB@
wait 60
expect_sysex @ERR_NOMODE_SYNTH@
expect mode 1
@MODE_SET_RUN4@
wait 200
expect mode 0
expect connected 1
